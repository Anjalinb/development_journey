### Add user:
git config --global user.name "your_name"

### add email:
git config --global user.email "your_email"

### list config details
git config --list

### command for initializing git
git init

## command for adding changed file to staging area
git add filename


## command for adding all changes to staging area
git add -A

### command for adding staged file to local repository
git commit -m "message"

### command for working tree clean or list all untracked files
git status

### command for adding remote repo to local repo
git remote add name path

### command for adding changed file to github
git push name master
